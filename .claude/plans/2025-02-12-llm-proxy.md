```go
// internal/proxy/models/costs.go
package models

// CalculateCost computes the cost for a given model and token usage
func CalculateCost(cfg *ModelConfig, inputTokens, outputTokens int) float64 {
    inputCost := float64(inputTokens) * cfg.InputCostPerMillion / 1000000
    outputCost := float64(outputTokens) * cfg.OutputCostPerMillion / 1000000
    return inputCost + outputCost
}

// EstimateCost calculates estimated cost from token estimates
func EstimateCost(cfg *ModelConfig, estimate *TokenEstimate) *CostEstimate {
    inputCost := float64(estimate.Input) * cfg.InputCostPerMillion / 1000000
    outputCost := float64(estimate.Output) * cfg.OutputCostPerMillion / 1000000

    return &CostEstimate{
        Input:  inputCost,
        Output: outputCost,
        Total:   inputCost + outputCost,
    }
}
```

```go
// internal/proxy/models/registry.go
package models

import (
    "sync"
    "gopkg.in/yaml.v3"
    "os"
)

// Registry holds all registered models from config
type Registry struct {
    mu      sync.RWMutex
    models   map[string]*ModelConfig  // key: model name
    router   RouterConfig
}

// RouterConfig holds configuration for the router model
type RouterConfig struct {
    Model    string `yaml:"model"`
    Timeout  string `yaml:"timeout"`
}

var (
    globalRegistry *Registry
    once         sync.Once
)

// Get returns the global model registry
func Get() *Registry {
    once.Do(func() {
        globalRegistry = &Registry{
            models: make(map[string]*ModelConfig),
        }
    })
    return globalRegistry
}

// LoadFromYAML loads model and router configuration from YAML file
func (r *Registry) LoadFromYAML(path string) error {
    data, err := os.ReadFile(path)
    if err != nil {
        return err
    }

    var config struct {
        Models map[string]*ModelConfig `yaml:"models"`
        Router RouterConfig          `yaml:"router"`
    }

    if err := yaml.Unmarshal(data, &config); err != nil {
        return err
    }

    r.mu.Lock()
    defer r.mu.Unlock()

    // Store models
    for name, cfg := range config.Models {
        r.models[name] = cfg
    }

    // Store router config
    r.router = config.Router

    return nil
}

// GetModel returns a model configuration by name
func (r *Registry) GetModel(name string) (*ModelConfig, bool) {
    r.mu.RLock()
    defer r.mu.RUnlock()

    cfg, exists := r.models[name]
    return cfg, exists
}

// GetRouterModel returns the configured router model name
func (r *Registry) GetRouterModel() string {
    r.mu.RLock()
    defer r.mu.RUnlock()

    return r.router.Model
}

// GetAllModels returns all registered model names
func (r *Registry) GetAllModels() []string {
    r.mu.RLock()
    defer r.mu.RUnlock()

    names := make([]string, 0, len(r.models))
    for name := range r.models {
        names = append(names, name)
    }
    return names
}

// GetModelsByProvider returns all models for a specific provider
func (r *Registry) GetModelsByProvider(provider string) []*ModelConfig {
    r.mu.RLock()
    defer r.mu.RUnlock()

    var models []*ModelConfig
    for _, cfg := range r.models {
        if cfg.Provider == provider {
            models = append(models, cfg)
        }
    }
    return models
}

// GetModelsByTier returns all models in a specific tier
func (r *Registry) GetModelsByTier(tier ModelTier) []*ModelConfig {
    r.mu.RLock()
    defer r.mu.RUnlock()

    var models []*ModelConfig
    for _, cfg := range r.models {
        if cfg.Tier == tier {
            models = append(models, cfg)
        }
    }
    return models
}
```

```go
// internal/proxy/quota/quota.go
package quota

import "context"

// CheckResult represents the result of a quota check
type CheckResult struct {
    HasQuota       bool
    Source         string  // "plan" or "addon"
    PlanPercent     int
    PlanDollars   float64
    AddonDollars  float64
    UsingAddon      bool
}

// CheckQuota verifies if user has sufficient quota for estimated cost
func CheckQuota(ctx context.Context, userID string, estimatedCost float64) (*CheckResult, error) {
    // Fetch from Redis:
    // - user:{id}:quota:plan_percent
    // - user:{id}:quota:plan_dollars
    // - user:{id}:quota:addon_dollars

    // Return CheckResult with:
    // - HasQuota: true if either plan or addon has sufficient funds
    // - Source: which will be charged
    // - UsingAddon: true if dipping into addon
}

// ReserveQuota reserves estimated cost before LLM call
func ReserveQuota(ctx context.Context, userID string, estimatedCost float64, source string) error {
    // Atomically:
    // INCRBY user:{id}:quota:reserved estimatedCost
    // DECRBY user:{id}:quota:plan_dollars estimatedCost
    // or DECRBY user:{id}:quota:addon_dollars estimatedCost
}

// DeductQuota calculates actual cost after stream and adjusts
func DeductQuota(ctx context.Context, userID string, reserved, actual float64, source string) error {
    // Calculate difference
    diff := reserved - actual

    // Adjust: return difference or charge more
    if diff > 0 {
        // Over-reserved, return some
        INCRBY user:{id}:quota:plan_dollars diff
    } else if diff < 0 {
        // Under-reserved, charge more
        DECRBY user:{id}:quota:plan_dollars -diff
    }

    // Release reservation
    DECRBY user:{id}:quota:reserved reserved
}
```

```go
    }

    // Parse JSON response
    var resp RouterResponse
```

```go
// internal/proxy/usage/writer.go
package usage

import (
    "context"
    "time"
    "github.com/redis/go-redis/v9"
)

const (
    batchSize     = 1000
    flushInterval = 30 * time.Second
    maxBufferSize = 10000
    maxRedisQueue = 100000
)

type UsageWriter struct {
    eventChan   chan proxy.UsageEvent
    buffer      []proxy.UsageEvent
    ticker      *time.Ticker
    db          *sql.DB
    redis       *redis.Client
    diskQueue   *DiskQueue
    dbHealthy   atomic.Bool
    batchSize   int
}

func NewUsageWriter(db *sql.DB, redis *redis.Client) *UsageWriter {
    return &UsageWriter{
        eventChan: make(chan proxy.UsageEvent, maxBufferSize),
        buffer:    make([]proxy.UsageEvent, 0, batchSize),
        ticker:    time.NewTicker(flushInterval),
        db:        db,
        redis:     redis,
        diskQueue: NewDiskQueue("/var/llm-gateway/usage-backup"),
        batchSize:  batchSize,
    }
}

func (w *UsageWriter) Start(ctx context.Context) {
    for {
        select {
        case event := <-w.eventChan:
            w.buffer = append(w.buffer, event)
            if len(w.buffer) >= w.batchSize {
                w.flush()
            }
        case <-w.ticker.C:
            if len(w.buffer) > 0 {
                w.flush()
            }
        case <-ctx.Done():
            w.flush() // Final flush
            return
        }
    }
}

func (w *UsageWriter) flush() error {
    if err := w.dbInsert(w.buffer); err != nil {
        w.dbHealthy.Set(false)
        // Move buffer to fallback
        for _, e := range w.buffer {
            w.writeWithFallback(e)
        }
        return err
    }
    w.dbHealthy.Set(true)
    w.buffer = nil
    w.recoverFallbackQueues()
    return nil
}

func (w *UsageWriter) writeWithFallback(event proxy.UsageEvent) {
    select {
    case w.eventChan <- event:
        return
    default:
        // Channel full, try Redis
        if err := w.pushToRedis(event); err == nil {
            return
        }
        // Redis failed, write to disk
        if err := w.diskQueue.Push(event); err != nil {
            // Last resort - alert
            alert.Critical("Usage tracking failure - data at risk", event)
        }
    }
}
```

```go
// internal/proxy/handler.go
package proxy

import (
    "context"
    "github.com/nevaria-ai/gateway/internal/agent"
    "github.com/nevaria-ai/gateway/internal/llm"
)

type Handler struct {
    router       *router.Router
    quota        *quota.Service
    health       *health.Service
    llmClient    *llm.Client
    usage        *usage.Writer
    models       *models.Config
}

type ProxyRequest struct {
    UserID    string
    SessionID string
    Query     string
    Tools     []agent.Tool
    Messages  []agent.AgentMessage
}

type ProxyResponse struct {
    Stream      <-chan agent.StreamEvent
    UsingAddon  bool
    AddonRemaining float64
}

// Handle processes a proxy request through the full flow
func (h *Handler) Handle(ctx context.Context, req ProxyRequest) (*ProxyResponse, error) {
    // 1. QUOTA CHECK
    checkResult, err := h.quota.Check(ctx, req.UserID)
    if err != nil {
        return nil, err
    }
    if !checkResult.HasQuota {
        return nil, ErrQuotaExceeded
    }

    // 2. ROUTER
    routerResp, err := h.router.Select(ctx, router.Request{
        UserQuery:   req.Query,
        UserContext:  checkResult.UserContext,
        // ...
    })
    if err != nil {
        return nil, err
    }

    // 3. HEALTH CHECK
    healthy, err := h.health.IsHealthy(ctx, routerResp.SelectedModel)
    if err != nil || !healthy {
        // Use fallback option
        routerResp.SelectedModel = routerResp.FallbackOptions[0]
    }

    // 4. RESERVE QUOTA
    err = h.quota.Reserve(ctx, req.UserID, routerResp.EstimatedCost.Total, checkResult.Source)
    if err != nil {
        return nil, err
    }

    // 5. LLM CALL
    stream, err := h.llmClient.Stream(ctx, routerResp.SelectedModel, req.Messages, req.Tools)

    // 6. Spawn goroutine to track usage after stream
    go h.trackStreamUsage(ctx, req.UserID, routerResp, stream)

    return &ProxyResponse{
        Stream:        stream,
        UsingAddon:    checkResult.UsingAddon,
        AddonRemaining: checkResult.AddonDollars,
    }, nil
}
```

```go
package proxy

import (
    "context"
    "github.com/nevaria-ai/gateway/internal/agent"
)

// ProxyClient implements agent.LLMClient with quota/routing/health
type ProxyClient struct {
    handler *Handler
    userID  string
}

// NewProxyClient creates a proxy client for a user
func NewProxyClient(handler *Handler, userID string) *ProxyClient {
    return &ProxyClient{
        handler: handler,
        userID:  userID,
    }
}

// Stream sends messages through the proxy flow
func (p *ProxyClient) Stream(ctx context.Context, messages []agent.AgentMessage, tools []agent.Tool) (<-chan agent.StreamEvent, error) {
    // Get last user message as query
    lastMsg := messages[len(messages)-1]

    req := ProxyRequest{
        UserID:   p.userID,
        Query:    extractText(lastMsg),
        Tools:    tools,
        Messages: messages,
    }

    resp, err := p.handler.Handle(ctx, req)
    if err != nil {
        return nil, err
    }

    // Add metadata to first event
    enhancedStream := make(chan agent.StreamEvent)

    go func() {
        defer close(enhancedStream)

        first := true
        for event := range resp.Stream {
            if first {
                // Add addon metadata
                event.UsingAddon = resp.UsingAddon
                first = false
            }
            enhancedStream <- event
        }
    }()

    return enhancedStream, nil
}
```
